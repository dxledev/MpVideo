cmake_minimum_required(VERSION 3.21)
project(MpVideo VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets OpenGLWidgets DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
set(THEME_SOURCES src/palette/theme.cpp src/palette/theme_source.cpp src/palette/theme_manager.cpp)
add_library(mpvideo_theme ${THEME_SOURCES} src/palette/theme_manager.h src/cli.cpp)
target_include_directories(mpvideo_theme PUBLIC src)
target_link_libraries(mpvideo_theme PUBLIC Qt6::Core Qt6::Gui)
add_library(mpvideo_player src/player.cpp src/video_widget.cpp)
target_include_directories(mpvideo_player PUBLIC src)
target_link_libraries(mpvideo_player PUBLIC Qt6::OpenGLWidgets PkgConfig::MPV)
add_library(mpvideo_ui src/window.cpp src/modal.cpp src/tooltip.cpp src/desktop_file_picker.cpp)
target_link_libraries(mpvideo_ui PUBLIC mpvideo_theme mpvideo_player Qt6::DBus)
add_executable(mpvideo src/main.cpp)
target_link_libraries(mpvideo PRIVATE mpvideo_ui)
foreach(target mpvideo_theme mpvideo_player mpvideo_ui mpvideo)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
option(MPVIDEO_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(MPVIDEO_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  foreach(target mpvideo_theme mpvideo_player mpvideo_ui mpvideo)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/theme_tests.cpp")
    add_executable(theme_tests tests/theme_tests.cpp)
    target_link_libraries(theme_tests PRIVATE mpvideo_theme Qt6::Test)
    add_test(NAME theme COMMAND theme_tests)
  endif()
  add_executable(window_tests tests/window_tests.cpp)
  target_link_libraries(window_tests PRIVATE mpvideo_ui Qt6::Test)
  find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
  add_test(NAME window COMMAND ${DBUS_RUN_SESSION} --config-file ${CMAKE_CURRENT_SOURCE_DIR}/tests/dbus-session.conf -- $<TARGET_FILE:window_tests>)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/player_tests.cpp")
    add_executable(player_tests tests/player_tests.cpp)
    target_link_libraries(player_tests PRIVATE mpvideo_player Qt6::Test)
    add_test(NAME player COMMAND player_tests)
    set_tests_properties(player PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()
include(GNUInstallDirs)
install(TARGETS mpvideo RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/io.github.mpvideo.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.github.mpvideo.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)

install(FILES assets/OMARCHY-DINO-MIT.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/mpvideo)
